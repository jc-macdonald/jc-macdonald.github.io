  \foreach \col in {0,...,4} {
    \pgfmathtruncatemacro{\missing}{mod(\row*3+\col,4)}
    \ifnum\missing=0
      \filldraw[fill=white,draw=Muted!55,dashed] (.1+\col*.5,-.65-\row*.5) rectangle ++(.4,-.4);
    \else
      \filldraw[fill=Blue!35,draw=Blue!65] (.1+\col*.5,-.65-\row*.5) rectangle ++(.4,-.4);
    \fi
  }
}

\node[panel=Teal,minimum width=8cm,text width=7.25cm,minimum height=2.6cm] (model) at (4.4,-.35)
  {{\color{Teal}\bfseries\large Variational low-rank model}\\[7pt]Latent scores and loadings\\Bias, observation noise and ARD priors\\Approximate posterior covariances};
\draw[flow] (2.8,-1.6) -- (model.west);
\node[panel=Green,minimum width=5.7cm,text width=4.95cm,minimum height=2.6cm] (prediction) at (0,-4.5)
  {{\color{Green}\bfseries\large Predictions}\\[6pt]Reconstruction / transform\\Predictive means and variances};
\node[panel=Gold,minimum width=5.7cm,text width=4.95cm,minimum height=2.6cm] (assessment) at (6.7,-4.5)

\draw[draw=Muted,line width=1pt] (model.south) -- (outputs);
\draw[flow] (outputs) -| (prediction.north);
\draw[flow] (outputs) -| (assessment.north);
\node[note] at ($(prediction.south west)+(0,-.6)$) {Fitting uses observed entries. Uncertainty is a variational approximation. Coverage and rank recovery need checks for the intended data and observation assumptions.};
